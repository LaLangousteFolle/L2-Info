SELECT id_produit, nom, id_fournisseur, pays FROM techstor_produits NATURAL JOIN techstor_fournisseurs GROUP BY id_produit, pays;
SELECT techstor_clients.id_client, nom, prenom, ville FROM techstor_clients LEFT JOIN techstor_commandes ON techstor_clients.id_client=techstor_commandes ON techstor_clients.id_client=techstor_commandes.id_client WHERE date_commande IS NULL^J;
